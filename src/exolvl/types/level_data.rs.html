16/// The level data for an Exoracer level.
17#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
18#[derive(Clone, Debug, Default, Hash, PartialEq, Eq)]
19#[allow(clippy::struct_excessive_bools)]
20pub struct LevelData {
21    /// The UUID of the level.
22    pub level_id: Uuid,
23    /// The version of the level e.g. v1, v2, etc.
24    pub level_version: i32,
25    /// Whether this level is for the new level editor.
26    ///
27    /// If this is true, the level can be opened in the new level editor. Otherwise it's for the "legacy" editor.
28    /// This Field is presumably only useful in .level files, not in .exolvl ones. A mismatch with the corresponding `LocalLevel` field should be avoided.
29    pub nova_level: bool,
30    /// The tile ids for the "under decoration" layer.
31    pub under_decoration_tiles: Vec<i32>,
32    /// The tile ids for the "background decoration" layer.
33    pub background_decoration_tiles: Vec<i32>,
34    /// The tile ids for the terrain layer.
35    pub terrain_tiles: Vec<i32>,
36    /// The tile ids for the floating zone layer.
37    pub floating_zone_tiles: Vec<i32>,
38    /// The tile ids for the "object" layer.
39    pub object_tiles: Vec<i32>,
40    /// The tile ids for the "foreground decoration" layer.
41    pub foreground_decoration_tiles: Vec<i32>,
42    /// The objects in the level.
43    pub objects: Vec<Object>,
44    /// The layers in the level.
45    pub layers: Vec<Layer>,
46    /// The prefabs in the level.
47    pub prefabs: Vec<Prefab>,
48    /// The brushes in the level.
49    pub brushes: Vec<Brush>,
50    /// The patterns in the level.
51    pub patterns: Vec<Pattern>,
52    /// The color palettes in the level.
53    ///
54    /// This is only present in levels with version 17 or higher.
55    pub color_palette: Option<Vec<Color>>,
56    /// The author medal time for this level in milliseconds.
57    pub author_time: i64,
58    /// The author medal lap times for this level in milliseconds.
59    pub author_lap_times: Vec<i64>,
60    /// The silver medal time for this level in milliseconds.
61    pub silver_medal_time: i64,
62    /// The gold medal time for this level in milliseconds.
63    pub gold_medal_time: i64,
64    /// The number of laps in this level.
65    pub laps: i32,
66    /// Whether the camera should be centered while playing this level.
67    ///
68    /// This is mostly deprecated and should stay set to false.
69    pub center_camera: bool,
70    /// The scripts in the level.
71    ///
72    /// These are used in the legacy level editor.
73    pub scripts: Vec<i32>,
74    /// The "new" scripts in the level.
75    ///
76    /// These are the scripts that are used in the new level editor. As opposed to the `scripts` field which is for the legacy editor.
77    pub nova_scripts: Vec<NovaScript>,
78    /// All the global variables in the level.
79    pub global_variables: Vec<Variable>,
80    /// The theme name of the level.
81    pub theme: Theme,
82    /// The custom background color of the level.
83    pub custom_background_color: Color,
84
85    /// Unknown data.
86    pub(crate) unknown1: [u8; 4],
87    /// The following terrain related fields are all used when explicitly copying certain terrain data.
88    ///
89    /// The custom terrain pattern that can be pasted with the `color_paste` button if the recieving object has the `FillMode` set to `Pattern`.
90    pub custom_terrain_pattern_id: i32,
91    /// The tiling of that pattern.
92    pub custom_terrain_pattern_tiling: Vec2,
93    /// the offset of that pattern.
94    pub custom_terrain_pattern_offset: Vec2,
95    /// In the legacy editor: The custom terrain color of the level.
96    /// In the new editor: The color of the copied terrain.
97    pub custom_terrain_color: Color,
98    /// Not 100% sure of the use of this, presumably the replacement for the border color in the new editor.
99    /// Used when copying and pasting properties of terrain.
100    pub custom_terrain_secondary_color: Color,
101    /// The blend mode of the copied terrain.
102    pub custom_terrain_blend_mode: i32,
103    /// The custom terrain border color of the level.
104    pub custom_terrain_border_color: Color,
105    /// The thickness of the terrain border.
106    pub custom_terrain_border_thickness: OrderedFloat<f32>,
107    /// The corner radius of the terrain border.
108    pub custom_terrain_border_corner_radius: OrderedFloat<f32>,
109    /// Whether the copied terrain has round reflex angles or not (only visual).
110    pub custom_terrain_round_reflex_angles: bool,
111    /// Whether the copied terrain has a round collider or not (not visual).
112    pub custom_terrain_round_collider: bool,
113    /// The friction of the copied terrain.
114    pub custom_terrain_friction: OrderedFloat<f32>,
115    /// Whether the default music should be played or not.
116    pub default_music: bool,
117    /// The music ids for the level. The game randomly picks one of these to play each time.
118    pub music_ids: Vec<String>,
119    /// Whether the game lets the player change directions or not.
120    pub allow_direction_change: bool,
121    /// Whether replays are disabled or not.
122    ///
123    /// If this is true, the game won't upload replays on this level.
124    /// (unless you explicitly upload blank shells from the history section, that only contain the time you set without any replay data. Could be a bug).
125    pub disable_replays: bool,
126    /// Whether revive pads are disabled or not.
127    pub disable_revive_pads: bool,
128    /// Whether the start animation is disabled or not.
129    pub disable_start_animation: bool,
130    /// The gravity vector for this level.
131    pub gravity: Vec2,
132}

134impl LevelData {
135    #[must_use]
136    pub fn default_with_id(level_id: Uuid) -> Self {
137        Self {
138            level_id,
139            level_version: 1,
140            nova_level: true,
141            color_palette: Some(Vec::default()),
142            laps: 1,
143            default_music: true,
144            gravity: Vec2 {
145                x: OrderedFloat(0.0),
146                y: OrderedFloat(-75.0),
147            },
148            ..Default::default()
149        }
150    }
